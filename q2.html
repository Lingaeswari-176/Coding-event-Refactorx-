Question 1 – Longest Substring Without Repeating Characters

def longestSubstring(s):
    max_len = 0

    for i in range(len(s)):
        temp = ""
        for j in range(i, len(s)):
            if s[j] in temp:
                break
            else:
                temp = temp + s[j]
                if len(temp) > max_len:
                    max_len = len(temp)

    return max_len


text = "abcabcbb"
print(longestSubstring(text))

Question 2 – Job Scheduling to Maximize Profit

def schedule(jobs):
    total = 0
    maxd = 0

    for i in range(len(jobs)):
        if jobs[i][1] > maxd:
            maxd = jobs[i][1]

    slots = [-1] * maxd

    for i in range(len(jobs)):
        for j in range(i + 1, len(jobs)):
            if jobs[i][2] < jobs[j][2]:
                jobs[i], jobs[j] = jobs[j], jobs[i]

    for i in range(len(jobs)):
        d = jobs[i][1]
        while d > 0:
            if slots[d - 1] == -1:
                slots[d - 1] = jobs[i][0]
                total = total + jobs[i][2]
                break
            d = d - 1

    return total


jobs = [
    ('A', 2, 100),
    ('B', 1, 19),
    ('C', 2, 27),
    ('D', 1, 25),
    ('E', 3, 15)
]

print(schedule(jobs))
