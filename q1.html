Question 1 – Second Largest Number

n = int(input())

numbers = []

i = 0

while i < n:
    
    value = int(input())
    
    numbers.append(value)
    
    i = i + 1

largest = numbers[0]

for num in numbers:
    
    if num > largest:
        
        largest = num

second = None

for num in numbers:
    
    if num != largest:
        
        if second is None or num > second:
            
            second = num

print(second)

Question 2 – Count Vowels

text = input()

count = 0

vowels = ['a','e','i','o','u','A','E','I','O','U']

i = 0

while i < len(text):
    
    ch = text[i]
    
    if ch in vowels:
        
        count = count + 1
    
    i = i + 1

print(count)

Question 3 – Most Frequent Word

sentence = input()

words = sentence.split()

max_count = 0

result = ""

for word in words:
    
    count = 0
    
    for w in words:
        
        if w == word:
            
            count = count + 1
    
    if count > max_count:
        
        max_count = count
        
        result = word

print(result)
