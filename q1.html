Question 1 – Two Sum (Naive)

nums = [2, 7, 11, 15]
target = 9

answer = []

found = False

for i in range(len(nums)):

    for j in range(len(nums)):

        if i == j:
            continue

        total = nums[i] + nums[j]

        if total == target:

            answer.append(i)
            answer.append(j)

            found = True

            break

    if found:
        break

Question 2 – Longest Substring Without Repeating Characters

def longest_substring(s):

    maximum = 0

    for i in range(len(s)):

        current = ""

        for j in range(i, len(s)):

            duplicate = False

            for ch in current:

                if ch == s[j]:
                    duplicate = True
                    break

            if duplicate:
                break

            current += s[j]

            if len(current) > maximum:
                maximum = len(current)

    return maximum


print(longest_substring("abcabcbb"))

Question 3 – Merge Intervals

intervals = [[1,3],[2,6],[8,10],[15,18]]

changed = True

while changed:

    changed = False

    i = 0

    while i < len(intervals):

        j = i + 1

        while j < len(intervals):

            if intervals[i][1] >= intervals[j][0] and intervals[j][1] >= intervals[i][0]:

                start = min(intervals[i][0], intervals[j][0])
                end = max(intervals[i][1], intervals[j][1])

                intervals[i] = [start, end]

                intervals.pop(j)

                changed = True

            else:

                j += 1

        i += 1
