Question 1 – Second Largest Number

nums = []
nums.append(5)
nums.append(2)
nums.append(8)
nums.append(3)
nums.append(7)

largest = -999
second = -999

for i in nums:
    if i > largest:
        second = largest
        largest = i
    else:
        if i > second and i != largest:
            second = i

print(second)

Question 2 – Count Vowels

text = "programming"

count = 0

for ch in text:
    if ch == 'a':
        count = count + 1
    elif ch == 'e':
        count = count + 1
    elif ch == 'i':
        count = count + 1
    elif ch == 'o':
        count = count + 1
    elif ch == 'u':
        count = count + 1

print(count)

Question 3 – Most Frequent Word

words = "apple banana apple orange apple".split()

max_count = 0
result = ""

for word in words:
    count = 0
    for w in words:
        if w == word:
            count = count + 1

    if count > max_count:
        max_count = count
        result = word

print(result)
